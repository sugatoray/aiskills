// index.html: every frame is a pure function of t
const OB = [0, 4, 9, 15, 22, 27, 32];  // authored scenes
const NB = OB.slice();               // retimed to the voice

function warp(tt) {
  for (let i = 1; i < NB.length; i++)
    if (tt <= NB[i]) return lerp(OB[i-1], OB[i],
      (tt - NB[i-1]) / (NB[i] - NB[i-1]));
}

function draw(t) {
  // camera: tilt, zoom, scroll
  cam.style.transform =
    `rotateY(${ry}deg) scale(${sc})`;
  // typing effect
  cmd.textContent = typed(line, t, a, b);
}

window.setT = (tt) => draw(warp(tt));
